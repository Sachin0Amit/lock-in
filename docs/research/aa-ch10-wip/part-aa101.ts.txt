/* (b) the x + 1/x ladder: each rung born from the two below it */
function in101LadderSvg(): string {
  const W = 640, H = 288
  const rungs = [3, 7, 18, 47, 123]
  const scale = 300 / 123
  const x0 = 150, barH = 24, top = 52, dy = 46
  const y = (i: number) => top + i * dy
  const w = (v: number) => 46 + v * scale
  const nodeX = 484
  let out: string[] = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="The ladder of symmetric sums: each rung s(k+1) is born from the two rungs below it by the recurrence.">`,
    in101ArrowDef(),
    in101Txt(20, 28, 'x + 1/x = 3', 13.5, C.mid, 'start', '600'),
    in101Txt(nodeX + 60, 28, 's(k+1) = 3·s(k) − s(k−1)', 12.5, C.conn, 'middle', '600'),
  ]
  rungs.forEach((v, i) => {
    out.push(`<rect x="${x0}" y="${y(i)}" width="${(w(v)).toFixed(1)}" height="${barH}" rx="5" fill="${C.teal}" stroke="${C.tealInk}" stroke-width="1.2"/>`)
    out.push(in101Txt(x0 - 12, y(i) + 17, `s${i + 1} = ${v}`, 14, C.ink, 'end', '600'))
    out.push(in101Txt(x0 + w(v) + 12, y(i) + 17, `${v}`, 13, C.tealInk, 'start'))
  })
  for (let k = 1; k <= 3; k++) {
    const srcA = { x: x0 + w(rungs[k - 1]) + 6, y: y(k - 1) + barH / 2 }
    const srcB = { x: x0 + w(rungs[k]) + 6, y: y(k) + barH / 2 }
    const tgt = { x: x0 - 8, y: y(k + 1) + barH / 2 }
    out.push(`<path d="M ${srcA.x.toFixed(1)} ${srcA.y} H ${nodeX} V ${tgt.y}" stroke="${C.conn}" stroke-width="1.4" fill="none"/>`)
    out.push(`<path d="M ${srcB.x.toFixed(1)} ${srcB.y} H ${nodeX}" stroke="${C.conn}" stroke-width="1.4" fill="none"/>`)
    out.push(`<circle cx="${nodeX}" cy="${tgt.y}" r="3.2" fill="${C.accent}"/>`)
    out.push(`<path d="M ${nodeX} ${tgt.y} H ${tgt.x}" stroke="${C.conn}" stroke-width="1.4" marker-end="url(#aa101Arrow)" fill="none"/>`)
  }
  out.push(in101Txt(W / 2, H - 10, 'two rungs below are all a new rung ever needs', 12, C.mid))
  out.push(`</svg>`)
  return out.join('')
}

/* (c) the AM–GM boundary: x(6−x) under its 9-ceiling with the equality point */
function in101AmgmSvg(): string {
  const W = 640, H = 304
  const ox = 66, oy = 254
  const xmax = 6.4, ymax = 10.4
  const pxPerX = (640 - ox - 26) / xmax
  const pyPerY = (oy - 44) / ymax
  const X = (x: number) => ox + x * pxPerX
  const Y = (y: number) => oy - y * pyPerY
  let pts = ''
  for (let x = 0; x <= 6.001; x += 0.05) pts += `${X(x).toFixed(1)},${Y(x * (6 - x)).toFixed(1)} `
  let out: string[] = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="The parabola y = x(6 − x) sits under the dashed ceiling y = 9 and touches it only at the equality point x = 3.">`,
    `<rect x="${X(0)}" y="${Y(9)}" width="${(X(6) - X(0)).toFixed(1)}" height="${(oy - Y(9)).toFixed(1)}" fill="${C.panelB}" opacity="0.55"/>`,
  ]
  for (let gx = 1; gx <= 6; gx++) out.push(`<line x1="${X(gx).toFixed(1)}" y1="${Y(9.9).toFixed(1)}" x2="${X(gx).toFixed(1)}" y2="${oy}" stroke="${C.faint}" stroke-width="1" opacity="0.4"/>`)

  return out.join('')

/* (d) the log-ordering chain as a decreasing staircase */
function in101LogStairSvg(): string {
  const W = 640, H = 272
  const vals = [1.5849625, 1.2618595, 1.1609640, 1.1132828, 1.0860331]
  const show = ['1.5850', '1.2619', '1.1610', '1.1133', '1.0860']
  const names = ['log2 3', 'log3 4', 'log4 5', 'log5 6', 'log6 7']
  const base = 212, bw = 76, gap = 22, x0 = 60, hScale = 116
  let out: string[] = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="A decreasing staircase of logarithms: log base n of n plus one falls strictly from 1.5850 down to 1.0860.">`,
    in101Txt(60, 28, 'add 1 to base and argument — the value still falls', 12.5, C.mid, 'start', '600'),
  ]
  const tops: string[] = []
  vals.forEach((v, i) => {
    const x = x0 + i * (bw + gap)
    const h = v * hScale
    const yTop = base - h
    out.push(`<rect x="${x}" y="${yTop.toFixed(1)}" width="${bw}" height="${h.toFixed(1)}" rx="6" fill="${C.teal}" stroke="${C.tealInk}" stroke-width="1.2"/>`)
    out.push(in101Txt(x + bw / 2, yTop - 8, show[i], 12.5, C.ink, 'middle', '600'))
    out.push(in101Txt(x + bw / 2, base + 20, names[i], 12.5, C.mid))
    tops.push(`${(x + bw / 2).toFixed(1)},${yTop.toFixed(1)}`)
    out.push(`<circle cx="${(x + bw / 2).toFixed(1)}" cy="${yTop.toFixed(1)}" r="3" fill="${C.accent}"/>`)
  })
  out.push(`<polyline points="${tops.join(' ')}" fill="none" stroke="${C.accent}" stroke-width="1.8" stroke-dasharray="6 4"/>`)
  out.push(`<line x1="${x0 - 14}" y1="${base}" x2="${x0 + 5 * bw + 4 * gap + 14}" y2="${base}" stroke="${C.struct}" stroke-width="1.6"/>`)
  out.push(in101Txt(W / 2, H - 10, 'strictly decreasing: 1.5850 → 1.2619 → 1.1610 → 1.1133 → 1.0860', 12, C.mid))
  out.push(`</svg>`)
  return out.join('')
}

/* solution figure S1: the four roots of exercise 1 on one number line */
function in101Ex1Svg(): string {
  const W = 640, H = 152, y0 = 86
  const xa = 44, xb = 596, lo = -0.25, hi = 3.45
  const px = (x: number) => xa + ((x - lo) / (hi - lo)) * (xb - xa)
  const rootsT3 = [0, 3], rootsT2 = [(3 - Math.sqrt(5)) / 2, (3 + Math.sqrt(5)) / 2]
  let out: string[] = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="Number line: the t equals 3 branch gives roots 0 and 3, the t equals 2 branch gives the surd pair near 0.382 and 2.618.">`,
    in101Txt(44, 26, '● t = 3 branch → x = 0, 3', 12.5, C.accent, 'start', '600'),
    in101Txt(44, 46, '● t = 2 branch → x = (3 ± √5)/2', 12.5, C.tealInk, 'start', '600'),
    `<line x1="${xa}" y1="${y0}" x2="${xb + 10}" y2="${y0}" stroke="${C.struct}" stroke-width="1.6"/>`,
    `<path d="M ${xb + 10} ${y0} l -8 -4 l 0 8 z" fill="${C.struct}"/>`,
  ]
  for (const t of [0, 1, 2, 3]) {
    out.push(`<line x1="${px(t).toFixed(1)}" y1="${y0}" x2="${px(t).toFixed(1)}" y2="${y0 + 6}" stroke="${C.conn}" stroke-width="1.3"/>`)
    out.push(in101Txt(px(t), y0 + 22, `${t}`, 11.5, C.mid))
  }
  for (const r of rootsT3) out.push(`<circle cx="${px(r).toFixed(1)}" cy="${y0}" r="5.5" fill="${C.accent}" stroke="${C.white}" stroke-width="1.3"/>`)
  for (const r of rootsT2) {
    out.push(`<circle cx="${px(r).toFixed(1)}" cy="${y0}" r="5.5" fill="${C.teal}" stroke="${C.white}" stroke-width="1.3"/>`)

  }
  out.push(in101Txt(px(rootsT2[0]) + 6, y0 + 44, '(3 − √5)/2 ≈ 0.382', 11.5, C.tealInk, 'start', '600'))
  out.push(in101Txt(px(rootsT2[1]) + 6, y0 + 44, '(3 + √5)/2 ≈ 2.618', 11.5, C.tealInk, 'start', '600'))
  out.push(`</svg>`)
  return out.join('')
}

/* solution figure S3: the log duel 5^40 vs 4^50 (exercise 13) */
function in101Ex13Svg(): string {
  const W = 640, H = 288, base = 222
  const bars = [
    { x: 190, w: 108, v: 64.3775165, top: '64.3775', lab: 'ln(5^40) = 40 ln 5', col: C.conn },
    { x: 372, w: 108, v: 69.3147181, top: '69.3147', lab: 'ln(4^50) = 50 ln 4', col: C.teal },
  ]
  const hScale = 1.92
  let out: string[] = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" role="img" aria-label="Two bars comparing 40 ln 5 against 50 ln 4: the taller bar for 50 ln 4 certifies that 4^50 beats 5^40.">`,
    in101Txt(320, 30, 'the taller log wins the duel', 13, C.mid, 'middle', '600'),
  ]
  for (const b of bars) {
    const h = b.v * hScale
    out.push(`<rect x="${b.x}" y="${(base - h).toFixed(1)}" width="${b.w}" height="${h.toFixed(1)}" rx="6" fill="${b.col}" stroke="${C.tealInk}" stroke-width="1.2"/>`)
    out.push(in101Txt(b.x + b.w / 2, base - h - 9, b.top, 13, C.ink, 'middle', '600'))
    out.push(in101Txt(b.x + b.w / 2, base + 22, b.lab, 12.5, C.mid))
  }
  out.push(`<line x1="${bars[0].x + bars[0].w}" y1="${(base - bars[0].v * hScale).toFixed(1)}" x2="${bars[1].x}" y2="${(base - bars[1].v * hScale).toFixed(1)}" stroke="${C.accent}" stroke-width="1.6" stroke-dasharray="5 4"/>`)
  out.push(`<line x1="150" y1="${base}" x2="530" y2="${base}" stroke="${C.struct}" stroke-width="1.6"/>`)
  out.push(in101Txt(320, H - 34, 'integer check: (4^5)^10 = 1024^10 beats (5^4)^10 = 625^10', 12.5, C.tealInk, 'middle', '600'))
  out.push(in101Txt(320, H - 12, 'same tenth-power read, no logarithm needed at all', 12, C.mid))
  out.push(`</svg>`)
  return out.join('')
}
